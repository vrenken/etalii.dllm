cmake_minimum_required(VERSION 3.18...3.31)
project(${SKBUILD_PROJECT_NAME} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Python 3.11 COMPONENTS Interpreter Development.Module REQUIRED)
find_package(nanobind CONFIG REQUIRED)

# NOMINSIZE: nanobind would otherwise add -Os after -O3 and build the kernels for size, at up to half the speed.
# The optimisation level never changes a bit; the floating point flags below fix the semantics.
nanobind_add_module(_kernels NB_STATIC NOMINSIZE cpp/kernels.cpp)
target_include_directories(_kernels PRIVATE cpp/include ${CMAKE_CURRENT_BINARY_DIR}/generated)

# The CUDA backend compiles its kernels at run time (NVRTC, see cpp/include/dllm/cuda.hpp), so their sources are
# embedded in the extension as byte arrays. Nothing CUDA is needed at build time.
function(dllm_embed_source variable path)
  file(READ "${path}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  set(${variable} "${bytes}0x00" PARENT_SCOPE)
endfunction()
set(DLLM_CUDA_SOURCES cpp/cuda/kernels.cu cpp/include/dllm/math.hpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${DLLM_CUDA_SOURCES})
dllm_embed_source(DLLM_CUDA_KERNELS_BYTES cpp/cuda/kernels.cu)
dllm_embed_source(DLLM_CUDA_MATH_BYTES cpp/include/dllm/math.hpp)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/dllm_cuda_sources.hpp"
  "// Generated by CMakeLists.txt from cpp/cuda/kernels.cu and cpp/include/dllm/math.hpp.\n"
  "#pragma once\n"
  "namespace dllm::cuda {\n"
  "inline constexpr unsigned char kCudaKernelsBytes[] = {${DLLM_CUDA_KERNELS_BYTES}};\n"
  "inline constexpr unsigned char kCudaMathBytes[] = {${DLLM_CUDA_MATH_BYTES}};\n"
  "inline const char* const kCudaKernelsSource = reinterpret_cast<const char*>(kCudaKernelsBytes);\n"
  "inline const char* const kCudaMathSource = reinterpret_cast<const char*>(kCudaMathBytes);\n"
  "}  // namespace dllm::cuda\n")
if(NOT WIN32)
  target_link_libraries(_kernels PRIVATE ${CMAKE_DL_LIBS})
endif()

# Determinism: never let the compiler reassociate or fuse floating point operations.
if(MSVC)
  # VS 2022 only contracts to FMA when /fp:contract is given explicitly.
  target_compile_options(_kernels PRIVATE /fp:precise)
else()
  target_compile_options(_kernels PRIVATE -ffp-contract=off -fno-fast-math)
endif()

install(TARGETS _kernels LIBRARY DESTINATION etalii_dllm)
